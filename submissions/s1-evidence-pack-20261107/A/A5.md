```python
#!/usr/bin/env python3
"""C-WORKER-FIXTURE-001 检测器：检查本脚本所在目录四项 fixture 齐备。

齐备(status_tool.py, test_status_tool.py, sample_fixed.json, README.md) → exit 0
任一缺失或不可读 → stderr 列出缺失项, exit 3
stdlib only；不联网、不执行任何清单命令。
"""
import sys
from pathlib import Path

REQUIRED = ("status_tool.py", "test_status_tool.py", "sample_fixed.json", "README.md")


def main(directory=None):
    base = Path(directory) if directory else Path(__file__).resolve().parent
    missing = [name for name in REQUIRED if not (base / name).is_file()]
    if missing:
        print(f"missing fixtures: {', '.join(missing)}", file=sys.stderr)
        return 3
    print("all fixtures present:", ", ".join(REQUIRED))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1] if len(sys.argv) > 1 else None))
```

```python
#!/usr/bin/env python3
"""红绿测试 for preflight.py：临时目录中缺 sample_fixed.json → exit 3；完整 → exit 0。"""
import subprocess
import sys
import tempfile
import unittest
from pathlib import Path

SCRIPT = Path(__file__).resolve().parent / "preflight.py"
FIXTURES = ("status_tool.py", "test_status_tool.py", "sample_fixed.json", "README.md")


def run_in(dirname):
    return subprocess.run([sys.executable, str(SCRIPT), dirname],
                          capture_output=True, text=True)


class TestPreflight(unittest.TestCase):
    def test_red_missing_sample_fixed(self):
        with tempfile.TemporaryDirectory() as d:
            for name in FIXTURES:
                if name != "sample_fixed.json":
                    (Path(d) / name).write_text("x")
            p = run_in(d)
            self.assertEqual(p.returncode, 3)
            self.assertIn("sample_fixed.json", p.stderr)

    def test_green_all_present(self):
        with tempfile.TemporaryDirectory() as d:
            for name in FIXTURES:
                (Path(d) / name).write_text("x")
            p = run_in(d)
            self.assertEqual(p.returncode, 0)

    def test_red_empty_dir(self):
        with tempfile.TemporaryDirectory() as d:
            p = run_in(d)
            self.assertEqual(p.returncode, 3)


if __name__ == "__main__":
    unittest.main(verbosity=2)
```
